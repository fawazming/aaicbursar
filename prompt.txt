function drawReceiptData(ctx, data) {
  // Loop through defined coordinates and render text dynamically
  for (const [key, field] of Object.entries(receiptFields.fields)) {
    if (data[key] !== undefined) {
      ctx.font = `${field.fontSize}px ${field.fontFamily}`;
      ctx.fillStyle = "#000000";
      ctx.textAlign = field.textAlign;
      ctx.fillText(data[key], field.x, field.y);
    }
  }
}
